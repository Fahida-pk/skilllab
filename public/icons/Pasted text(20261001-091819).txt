/* Convert UI time values like "10:03 PM" or "06:00 AM"
   into MySQL TIME values like "22:03:00" / "06:00:00". */
function normalizeDbTime($value) {
    $value = trim((string)($value ?? ""));
    if ($value === "") return null;

    if (preg_match('/^(\d{1,2}):(\d{2})(?::(\d{2}))?$/', $value, $m)) {
        $hour = (int)$m[1];
        $minute = (int)$m[2];
        $second = isset($m[3]) ? (int)$m[3] : 0;
        if ($hour >= 0 && $hour <= 23 && $minute >= 0 && $minute <= 59 && $second >= 0 && $second <= 59) {
            return sprintf("%02d:%02d:%02d", $hour, $minute, $second);
        }
    }

    if (preg_match('/^(\d{1,2}):(\d{2})\s*(AM|PM)$/i', $value, $m)) {
        $hour = (int)$m[1];
        $minute = (int)$m[2];
        $ampm = strtoupper($m[3]);
        if ($hour < 1 || $hour > 12 || $minute < 0 || $minute > 59) return null;
        if ($ampm === "AM" && $hour === 12) $hour = 0;
        if ($ampm === "PM" && $hour !== 12) $hour += 12;
        return sprintf("%02d:%02d:00", $hour, $minute);
    }

    $timestamp = strtotime($value);
    if ($timestamp === false) return null;
    return date("H:i:s", $timestamp);
}

function saveTaskImage($file, $uploadDir, $baseUrl)
{
    if (!$file || !isset($file['tmp_name']) || $file['error'] !== UPLOAD_ERR_OK) {
        return [
            "success" => false,
            "message" => "No valid image uploaded"
        ];
    }

    if ((int)$file['size'] > 5 * 1024 * 1024) {
        return [
            "success" => false,
            "message" => "Image must be 5 MB or smaller"
        ];
    }

    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mime = $finfo->file($file['tmp_name']);

    $allowed = [
        "image/jpeg" => "jpg",
        "image/png"  => "png",
        "image/gif"  => "gif",
        "image/webp" => "webp"
    ];

    if (!isset($allowed[$mime])) {
        return [
            "success" => false,
            "message" => "Only JPG, PNG, GIF or WEBP images are allowed"
        ];
    }

    $filename = "task_" . bin2hex(random_bytes(12)) . "." . $allowed[$mime];
    $target = $uploadDir . $filename;

    if (!move_uploaded_file($file['tmp_name'], $target)) {
        return [
            "success" => false,
            "message" => "Could not save uploaded image"
        ];
    }

    return [
        "success" => true,
        "filename" => $filename,
        "url" => $baseUrl . rawurlencode($filename)
    ];
}

/* =========================================
   STATUS HELPER
========================================= */

function getTaskStatus($task, $selectedDate)
{
    /*
        Already completed
    */
    if ((int)$task['status'] === 1) {
        return "completed";
    }

    $taskDate = $task['task_date'];

    /*
        Future
    */
    if ($taskDate > $selectedDate) {
        return "not_started";
    }

    /*
        Previous date
    */
    if ($taskDate < $selectedDate) {
        return "pending";
    }

    /*
        Selected date
    */

    if (!$task['from_time']) {
        return "not_started";
    }

    $nowMinutes =
        ((int)date("H") * 60) +
        (int)date("i");

    $from = strtotime($task['from_time']);

    if (!$from) {
        return "not_started";
    }

    $fromMinutes =
        ((int)date("H", $from) * 60) +
        (int)date("i", $from);


    /*
        No ending time
    */

    if (!$task['to_time']) {

        if ($nowMinutes >= $fromMinutes) {
            return "in_progress";
        }

        return "not_started";
    }


    $to = strtotime($task['to_time']);

    if (!$to) {
        return "not_started";
    }

    $toMinutes =
        ((int)date("H", $to) * 60) +
        (int)date("i", $to);


    /*
        Normal task
        5 AM - 10 AM
    */

    if ($toMinutes >= $fromMinutes) {

        if (
            $nowMinutes >= $fromMinutes &&
            $nowMinutes <= $toMinutes
        ) {
            return "in_progress";
        }

        if ($nowMinutes > $toMinutes) {
            return "pending";
        }

        return "not_started";
    }


    /*
        Overnight
        10 PM - 8 AM
    */

    if (
        $nowMinutes >= $fromMinutes ||
        $nowMinutes <= $toMinutes
    ) {
        return "in_progress";
    }

    return "pending";
}


/* =========================================
   PERCENTAGE
========================================= */

function percentage($completed, $total)
{
    if ($total <= 0) {
        return 0;
    }

    return round(
        ($completed / $total) * 100
    );
}

if ($action === "ensure_defaults") {

                'to' => $r['to_time'] ?? '',

        while ($r = mysqli_fetch_assoc($dateOverrideResult)) {
            $id = (string)$r['default_id'];
            if ((int)$r['is_active'] !== 1) {
                unset($serverDefaults[$id]);

            $serverDefaults[$id] = [
                'default_id' => $id,
                'title' => $r['task_title'],
                'from' => $r['from_time'],

    $todayWakeExplicit = false;

    if ($todayWakeDeleted) {
        $todayWakeDeleted = true;
        unset($serverDefaults['d1']);

    if (!$todayWakeExplicit && !$todayWakeDeleted) {

        $titleRaw = trim((string)($task['title'] ?? ''));

/* =========================================
   TASK TIME CONFLICT HELPERS
   -----------------------------------------
   One real-world time slot can belong to only
   one task for a student.

   Important date rule:
   - A normal task belongs entirely to task_date.
   - An overnight task (example 10:30 PM -> 5:00 AM)
     belongs to task_date from 10:30 PM -> midnight,
     and to the NEXT calendar date from midnight -> 5:00 AM.
   - Therefore a task on 30-Sep at 2:00 AM must also
     be checked against a 29-Sep overnight Sleep.
========================================= */

function dbTimeToMinutes($value)
{
    if ($value === null) return null;

    $value = trim((string)$value);

    if ($value === '') return null;

    /*
     * Always normalize first.
     * This supports BOTH:
     *   22:00:00 / 06:00:00
     *   10:00 PM / 06:00 AM
     *
     * This is important because conflict checking must understand
     * the AM/PM part. For example:
     *   1:00 PM = 13:00 = 780 minutes
     *   4:00 PM = 16:00 = 960 minutes
     *
     * Treating "1:00 PM" as just "1:00" would incorrectly make it
     * 1:00 AM and could falsely conflict with an overnight Sleep.
     */
    $normalized = normalizeDbTime($value);

    if ($normalized === null) {
        return null;
    }

    $parts = explode(':', $normalized);

    if (count($parts) < 2) return null;

    return ((int)$parts[0] * 60) + (int)$parts[1];
}

function taskRangesForSameDate($fromDb, $toDb)
{
    $from = dbTimeToMinutes($fromDb);

    if ($from === null) {
        return [];
    }

    // Point-time task.
    if ($toDb === null || $toDb === "") {
        return [
            ["start" => $from, "end" => $from]
        ];
    }

    $to = dbTimeToMinutes($toDb);

    if ($to === null) {
        return [];
    }

    // Same-day task.
    if ($to > $from) {
        return [
            ["start" => $from, "end" => $to]
        ];
    }

    // Overnight task: only the portion before midnight
    // belongs to its own task_date.
    return [
        ["start" => $from, "end" => 1440]
    ];
}

function taskRangesForPreviousDateTail($fromDb, $toDb)
{
    $from = dbTimeToMinutes($fromDb);
    $to = dbTimeToMinutes($toDb);

    if ($from === null || $to === null) {
        return [];
    }

    // Only an overnight task has a portion on the next date.
    if ($to > $from) {
        return [];
    }

    return [
        ["start" => 0, "end" => $to]
    ];
}

function taskRangesOverlap($a, $b)
{
    $aPoint = $a["start"] === $a["end"];
    $bPoint = $b["start"] === $b["end"];

    // Two point-time tasks at exactly the same time are a conflict.
    if ($aPoint && $bPoint) {
        return $a["start"] === $b["start"];
    }

    // Point task against interval.
    if ($aPoint) {
        return $b["start"] <= $a["start"]
            && $a["start"] < $b["end"];
    }

    if ($bPoint) {
        return $a["start"] <= $b["start"]
            && $b["start"] < $a["end"];
    }

    return (
        $a["start"] < $b["end"] &&
        $b["start"] < $a["end"]
    );
}

/*
 * Check the complete real-world time interval against:
 *   1. Other tasks on the same task_date.
 *   2. Previous-date overnight tasks.
 *   3. Next-date tasks when the candidate itself is overnight.
 *
 * $excludeId is used while editing so the task does not conflict
 * with itself.
 */
function findTaskTimeConflict(
    $conn,
    $userId,
    $taskDate,
    $fromDb,
    $toDb,
    $excludeId = null,
    $ignoreNextDayWakeUp = false,
    $candidateTitle = ""
) {
    $candidateFrom = dbTimeToMinutes($fromDb);
    $candidateTo = dbTimeToMinutes($toDb);

    if ($candidateFrom === null) {
        return null;
    }

    $candidateIsPoint =
        ($toDb === null || $toDb === "");

    $candidateOvernight =
        !$candidateIsPoint &&
        $candidateTo <= $candidateFrom;

    $candidateSameDateRanges = taskRangesForSameDate(
        $fromDb,
        $toDb
    );

    /*
     * ---------------------------------------------------------
     * 1. SAME DATE
     * ---------------------------------------------------------
     */
    $dateEsc = mysqli_real_escape_string(
        $conn,
        $taskDate
    );

    $excludeSql = "";

    if ($excludeId !== null && (int)$excludeId > 0) {
        $excludeSql =
            " AND t.id != " . (int)$excludeId;
    }

    $sameDateSql = "
        SELECT
            t.id,
            t.task_name,
            t.from_time,
            t.to_time
        FROM tasks t
        WHERE t.user_id=" . (int)$userId . "
          AND t.task_date='$dateEsc'
          $excludeSql
        ORDER BY t.from_time ASC, t.id ASC
    ";

    $sameDateResult = mysqli_query(
        $conn,
        $sameDateSql
    );

    if ($sameDateResult) {
        while ($existing = mysqli_fetch_assoc($sameDateResult)) {
            $existingRanges = taskRangesForSameDate(
                $existing["from_time"],
                $existing["to_time"]
            );

            foreach ($candidateSameDateRanges as $candidateRange) {
                foreach ($existingRanges as $existingRange) {
                    $candidatePoint =
                        $candidateRange["start"] === $candidateRange["end"];

                    $existingPoint =
                        $existingRange["start"] === $existingRange["end"];

                    $existingIsWakeUp =
                        strtolower(trim((string)($existing["task_name"] ?? ""))) === "wake up";

                    $candidateIsWakeUp =
                        strtolower(trim((string)$candidateTitle)) === "wake up";

                    // Wake Up is a point marker. It may share the exact
                    // start boundary of a duration task.
                    if (
                        $existingIsWakeUp &&
                        $existingPoint &&
                        !$candidatePoint &&
                        $candidateRange["start"] === $existingRange["start"]
                    ) {
                        continue;
                    }

                    if (
                        $candidateIsWakeUp &&
                        $candidatePoint &&
                        !$existingPoint &&
                        $candidateRange["start"] === $existingRange["start"]
                    ) {
                        continue;
                    }

                    if (taskRangesOverlap(
                        $candidateRange,
                        $existingRange
                    )) {
                        return $existing;
                    }
                }
            }
        }
    }

    /*
     * ---------------------------------------------------------
     * 2. PREVIOUS DATE -> CURRENT DATE
     *
     * Example:
     * 29-Sep Sleep 10:30 PM -> 5:00 AM
     * 30-Sep task 2:00 AM -> 3:00 AM
     *
     * This MUST be rejected.
     *
     * 30-Sep Wake Up 5:00 AM is allowed because Sleep ends
     * exactly at 5:00 AM.
     * ---------------------------------------------------------
     */
    $previousDate = date(
        "Y-m-d",
        strtotime($taskDate . " -1 day")
    );

    $previousDateEsc = mysqli_real_escape_string(
        $conn,
        $previousDate
    );

    $previousSql = "
        SELECT
            t.id,
            t.task_name,
            t.from_time,
            t.to_time
        FROM tasks t
        WHERE t.user_id=" . (int)$userId . "
          AND t.task_date='$previousDateEsc'
          AND t.to_time IS NOT NULL
          AND t.to_time <> ''
          AND t.to_time <= t.from_time
        ORDER BY t.from_time ASC, t.id ASC
    ";

    if ($excludeId !== null && (int)$excludeId > 0) {
        $previousSql =
            str_replace(
                "ORDER BY t.from_time ASC, t.id ASC",
                "AND t.id != " . (int)$excludeId . "
                 ORDER BY t.from_time ASC, t.id ASC",
                $previousSql
            );
    }

    $previousResult = mysqli_query(
        $conn,
        $previousSql
    );

    if ($previousResult) {
        while ($previousTask = mysqli_fetch_assoc($previousResult)) {
            $previousRanges =
                taskRangesForPreviousDateTail(
                    $previousTask["from_time"],
                    $previousTask["to_time"]
                );

            foreach ($candidateSameDateRanges as $candidateRange) {
                foreach ($previousRanges as $previousRange) {
                    if (taskRangesOverlap(
                        $candidateRange,
                        $previousRange
                    )) {
                        return $previousTask;
                    }
                }
            }
        }
    }

    /*
     * ---------------------------------------------------------
     * 3. CURRENT DATE OVERNIGHT -> NEXT DATE
     *
     * Example:
     * 29-Sep Sleep 10:30 PM -> 5:00 AM
     * 30-Sep task 2:00 AM -> 3:00 AM
     *
     * This is checked when the overnight task itself is being
     * added/edited.
     * ---------------------------------------------------------
     */
    if ($candidateOvernight) {
        $nextDate = date(
            "Y-m-d",
            strtotime($taskDate . " +1 day")
        );

        $nextDateEsc = mysqli_real_escape_string(
            $conn,
            $nextDate
        );

        $nextSql = "
            SELECT
                t.id,
                t.task_name,
                t.from_time,
                t.to_time
            FROM tasks t
            WHERE t.user_id=" . (int)$userId . "
              AND t.task_date='$nextDateEsc'
        ";

        // When editing overnight Sleep, the inherited next-day Wake Up is
        // updated to Sleep's new end time, so it must not block the edit.
        // Other next-day tasks remain real conflicts.
        if ($ignoreNextDayWakeUp) {
            $nextSql .= "
              AND NOT EXISTS (
                  SELECT 1
                  FROM skilllab_task_default_map wm
                  WHERE wm.task_id=t.id
                    AND wm.user_id=t.user_id
                    AND wm.task_date=t.task_date
                    AND wm.default_id='d1'
              )
            ";
        }

        $nextSql .= "
            ORDER BY t.from_time ASC, t.id ASC
        ";

        if ($excludeId !== null && (int)$excludeId > 0) {
            $nextSql =
                str_replace(
                    "ORDER BY t.from_time ASC, t.id ASC",
                    "AND t.id != " . (int)$excludeId . "
                     ORDER BY t.from_time ASC, t.id ASC",
                    $nextSql
                );
        }

        $nextResult = mysqli_query(
            $conn,
            $nextSql
        );

        if ($nextResult) {
            $candidateNextDateRange = [
                "start" => 0,
                "end" => $candidateTo
            ];

            while ($nextTask = mysqli_fetch_assoc($nextResult)) {
                $nextRanges = taskRangesForSameDate(
                    $nextTask["from_time"],
                    $nextTask["to_time"]
                );

                foreach ($nextRanges as $nextRange) {
                    if (taskRangesOverlap(
                        $candidateNextDateRange,
                        $nextRange
                    )) {
                        return $nextTask;
                    }
                }
            }
        }
    }

    return null;
}

function sendTaskTimeConflict($existingTask)
{
    $existingFrom = $existingTask["from_time"] ?? "";
    $existingTo = $existingTask["to_time"] ?? "";

    $existingTime = $existingTo
        ? $existingFrom . " - " . $existingTo
        : $existingFrom;

    echo json_encode([
        "success" => false,
        "message" =>
            "This time cannot be scheduled.\n\n" .
            ($existingTask["task_name"] ?? "Another task") .
            " is already scheduled for " .
            $existingTime .
            ".\n\nPlease choose another time."
    ]);

    exit;
}

    $sql = "

         AND m.task_date = t.task_date
        WHERE t.user_id='$user_id'
          AND t.task_date='$date'
        ORDER BY t.from_time ASC
    ";

        echo json_encode([
            "success" => false,
            "error" => mysqli_error($conn)
        ]);

        exit;

            "completed" =>
                ((int)$row['status'] === 1),

            "default_id" =>
                $row['mapped_default_id'] ?? null,

            "sleep_date"=>$sleepDate,

    /* Get the recurring Sleep definition so only the end time changes. */
    $baseSleep = $conn->prepare("
        SELECT task_title, from_time, icon, color
        FROM student_default_tasks
        WHERE user_id=? AND default_id='d5'

if ($action === "update") {

if ($action === "delete") {

    $defaultIdRaw = trim((string)($data['default_id'] ?? $data['defaultId'] ?? ''));
    $deleteDateRaw = trim((string)($data['task_date'] ?? $data['date'] ?? ''));
    $defaultIdEsc = mysqli_real_escape_string($conn, $defaultIdRaw);
    $deleteDateEsc = mysqli_real_escape_string(
        $conn,
        $deleteDateRaw ?: $today
    );


    if (!$id) {

        echo json_encode([
            "success" => false,
            "message" => "ID missing"
        ]);

        exit;
    }


    /* Check original task date */
    $checkSql = "
        SELECT task_date
        FROM tasks
        WHERE id='$id'
        AND user_id='$user_id'
        LIMIT 1
    ";

    $checkResult = mysqli_query($conn, $checkSql);

    if (
        !$checkResult ||
        !($checkRow = mysqli_fetch_assoc($checkResult))
    ) {

        echo json_encode([
            "success" => false,
            "message" => "Task not found"
        ]);

        exit;
    }


    $taskDate = $checkRow['task_date'];


    /* Previous day cannot be deleted */
    if ($taskDate < $today) {

        echo json_encode([
            "success" => false,
            "message" => "Previous day tasks cannot be deleted."
        ]);

        exit;
    }


    /*
     * EVERY deleted task gets a date-wise state. Older student-created
     * tasks may not have a default_id, so infer the mapping or create a
     * stable custom id from the task id.
     */
    if ($defaultIdRaw === '') {
        $mapStmt = $conn->prepare("
            SELECT default_id
            FROM skilllab_task_default_map
            WHERE task_id=? AND user_id=?
            LIMIT 1
        ");
        $taskIdInt = (int)$id;
        $mapStmt->bind_param("ii", $taskIdInt, $user_id);
        $mapStmt->execute();
        $mapRow = $mapStmt->get_result()->fetch_assoc();
        $mapStmt->close();

        $defaultIdRaw = ($mapRow && !empty($mapRow['default_id']))
            ? (string)$mapRow['default_id']
            : ('c_task_' . $taskIdInt);
        $defaultIdEsc = mysqli_real_escape_string($conn, $defaultIdRaw);
    }

    // Always use the task's real date for the date-wise state.
    // A delete on 29-Sep must never affect another date.
    $effectiveDeleteDate = $taskDate;

    $deleteMarkerSql = "
        INSERT INTO skilllab_deleted_default_tasks
            (user_id, task_date, default_id)
        VALUES
            ('$user_id', '$effectiveDeleteDate', '$defaultIdEsc')
        ON DUPLICATE KEY UPDATE created_at=CURRENT_TIMESTAMP
    ";

    if (!mysqli_query($conn, $deleteMarkerSql)) {
        echo json_encode([
            "success" => false,
            "message" => "Could not save date-wise delete state",
            "error" => mysqli_error($conn)
        ]);
        exit;
    }

    /* Keep task -> default identity for this exact date. */
    $mapDeleteSql = "
        INSERT INTO skilllab_task_default_map
            (task_id, user_id, task_date, default_id)
        VALUES
            ('$id', '$user_id', '$effectiveDeleteDate', '$defaultIdEsc')
        ON DUPLICATE KEY UPDATE
            task_date=VALUES(task_date),
            default_id=VALUES(default_id)
    ";

    if (!mysqli_query($conn, $mapDeleteSql)) {
        echo json_encode([
            "success" => false,
            "message" => "Could not save task mapping",
            "error" => mysqli_error($conn)
        ]);
        exit;
    }

    /* Date-wise inactive override. */
    $deleteOverride = $conn->prepare("
        INSERT INTO student_default_task_dates
            (user_id, task_date, default_id, task_title, from_time, to_time, is_active, source)
        VALUES (?, ?, ?, '', NULL, NULL, 0, 'student_deleted')

    $deleteOverride->bind_param("iss", $user_id, $effectiveDeleteDate, $defaultIdRaw);

    if (!$deleteOverride->execute()) {
        $err = $conn->error;
        $deleteOverride->close();
        echo json_encode([
            "success" => false,
            "message" => "Could not save inactive date override",
            "error" => $err
        ]);
        exit;
    }
    $deleteOverride->close();

    $sql = "
        DELETE FROM tasks
        WHERE id='$id'
        AND user_id='$user_id'
    ";


    if (mysqli_query($conn, $sql)) {

        echo json_encode([
            "success" => true
        ]);

    } else {

        echo json_encode([
            "success" => false,
            "error" => mysqli_error($conn)
        ]);
    }

    exit;

if ($action === "toggle") {

    $id = mysqli_real_escape_string(
        $conn,
        $data['id'] ?? ''
    );

    $status = mysqli_real_escape_string(
        $conn,
        $data['status'] ?? 0
    );


    if (!$id) {

        echo json_encode([
            "success" => false,
            "message" => "ID missing"
        ]);

        exit;
    }


    /* Check original task date + start time */
    $checkSql = "
        SELECT task_date, from_time
        FROM tasks
        WHERE id='$id'
        AND user_id='$user_id'
        LIMIT 1
    ";

    $checkResult = mysqli_query($conn, $checkSql);

    if (
        !$checkResult ||
        !($checkRow = mysqli_fetch_assoc($checkResult))
    ) {

        echo json_encode([
            "success" => false,
            "message" => "Task not found"
        ]);

        exit;
    }


    $taskDate = $checkRow['task_date'];
    $fromTime = $checkRow['from_time'] ?? '';


    /* ONLY TODAY can be ticked */
    if ($taskDate !== $today) {

        echo json_encode([
            "success" => false,
            "message" => "Only today's tasks can be completed."
        ]);

        exit;
    }


    /* =========================================================
       TASK ACCURACY
       ---------------------------------------------------------
       Tick within 15 minutes  -> 100%
       Tick within 1 hour      -> 50%
       Tick after 1 hour       -> 30%

       0% is only the unticked/default state.
       ========================================================= */

    $taskAccuracy = 0;

    /* Only calculate accuracy when task is being ticked. */
    if ((int)$status === 1 && $fromTime !== '') {

        $fromTimestamp = strtotime($fromTime);
        $nowTimestamp  = time();

        if ($fromTimestamp !== false) {

            $elapsedMinutes = floor(
                ($nowTimestamp - $fromTimestamp) / 60
            );

            /*
             * If the task is ticked before its start time,
             * do not allow a negative elapsed value.
             */
            $elapsedMinutes = max(0, $elapsedMinutes);


            if ($elapsedMinutes <= 15) {

                $taskAccuracy = 100;

            } elseif ($elapsedMinutes <= 60) {

                $taskAccuracy = 50;

            } else {

                $taskAccuracy = 30;
            }
        }
    }


    /* =========================================================
       SAVE STATUS + TASK ACCURACY
       ========================================================= */

    $sql = "
        UPDATE tasks
        SET
            status='$status',
            task_accuracy_percentage='$taskAccuracy'
        WHERE id='$id'
        AND user_id='$user_id'
    ";


    if (mysqli_query($conn, $sql)) {

        echo json_encode([
            "success" => true,
            "task_accuracy_percentage" => $taskAccuracy,
            "performance_time" => $taskAccuracy
        ]);

    } else {

        echo json_encode([
            "success" => false,
            "error" => mysqli_error($conn)
        ]);
    }

    exit;
}

    /* ===============================
       MONTH
    =============================== */

    $monthStart = clone $dateObj;

    $monthStart->modify(
        "first day of this month"
    );


    $monthEnd = clone $dateObj;

    $monthEnd->modify(
        "last day of this month"
    );


    $weekStartStr =
        $weekStart->format("Y-m-d");

    $weekEndStr =
        $weekEnd->format("Y-m-d");

    $monthStartStr =
        $monthStart->format("Y-m-d");

    $monthEndStr =
        $monthEnd->format("Y-m-d");


    /* ===============================
       GET MONTH TASKS
    =============================== */

    $sql = "
        SELECT
            id,
            task_name,
            task_date,
            from_time,
            to_time,
            status,
            task_percentage
        FROM tasks
        WHERE user_id='$user_id'
        AND task_date BETWEEN
            '$monthStartStr'
            AND
            '$monthEndStr'
        ORDER BY
            task_date ASC,
            from_time ASC
    ";


    $result = mysqli_query(
        $conn,
        $sql
    );


    if (!$result) {

        echo json_encode([
            "success" => false,
            "message" =>
                mysqli_error($conn)
        ]);

        exit;
    }


    $allTasks = [];


    while ($row = mysqli_fetch_assoc($result)) {

        $row['id'] =
            (int)$row['id'];

        $row['status'] =
            (int)$row['status'];


        $row['task_status'] =
            getTaskStatus(
                $row,
                $selectedDate
            );


        $allTasks[] = $row;
    }


    /* ===============================
       FILTER
    =============================== */

    $todayTasks = [];

    $weekTasks = [];


    foreach ($allTasks as $task) {

        if (
            $task['task_date'] ===
            $selectedDate
        ) {
            $todayTasks[] = $task;
        }


        if (
            $task['task_date'] >=
                $weekStartStr
            &&
            $task['task_date'] <=
                $weekEndStr
        ) {
            $weekTasks[] = $task;
        }
    }


    /* ===============================
       TODAY STATS
    =============================== */

    $todayTotal =
        count($todayTasks);

    $todayCompleted = 0;
    $todayInProgress = 0;
    $todayPending = 0;
    $todayNotStarted = 0;


    foreach ($todayTasks as $task) {

        switch (
            $task['task_status']
        ) {

            case "completed":

                $todayCompleted++;

                break;


            case "in_progress":

                $todayInProgress++;

                break;


            case "pending":

                $todayPending++;

                break;


            case "not_started":

                $todayNotStarted++;

                break;
        }
    }


    /* ===============================
       WEEK STATS
    =============================== */

    $weekTotal =
        count($weekTasks);

    $weekCompleted = 0;


    foreach ($weekTasks as $task) {

        if (
            $task['task_status'] ===
            "completed"
        ) {

            $weekCompleted++;
        }
    }


    /* ===============================
       MONTH STATS
    =============================== */

    $monthTotal =
        count($allTasks);

    $monthCompleted = 0;


    foreach ($allTasks as $task) {

        if (
            $task['task_status'] ===
            "completed"
        ) {

            $monthCompleted++;
        }
    }


    /* ===============================
       STUDY HOURS
    =============================== */

    $studyMinutes = 0;


    foreach ($weekTasks as $task) {

        if (
            !$task['from_time'] ||
            !$task['to_time']
        ) {
            continue;
        }


        $from =
            strtotime($task['from_time']);

        $to =
            strtotime($task['to_time']);


        if (!$from || !$to) {
            continue;
        }


        $minutes =
            ($to - $from) / 60;


        if ($minutes < 0) {
            $minutes += 1440;
        }


        if ($minutes > 0) {
            $studyMinutes += $minutes;
        }
    }


    $studyHours =
        floor($studyMinutes / 60);

    $studyRemainingMinutes =
        $studyMinutes % 60;


    /* ===============================
       FORMAT TODAY TASKS
    =============================== */

    $formattedTasks = [];


    foreach ($todayTasks as $task) {

        $formattedTasks[] = [

            "id" =>
                $task['id'],

            "title" =>
                $task['task_name'],

            "from" =>
                $task['from_time'],

            "to" =>
                $task['to_time'],

            "completed" =>
                ((int)$task['status'] === 1),

            "percentage" =>
                max(0, min(100, (int)($task['task_percentage'] ?? 0))),

            "taskStatus" =>
                $task['task_status']
        ];
    }


    /* ===============================
       DASHBOARD RESPONSE
    =============================== */

    echo json_encode([

        "success" => true,

        "date" =>
            $selectedDate,


        /* TODAY */

        "today" => [

            "total" =>
                $todayTotal,

            "completed" =>
                $todayCompleted,

            "inProgress" =>
                $todayInProgress,

            "pending" =>
                $todayPending,

            "notStarted" =>
                $todayNotStarted,

            "percentage" =>
                percentage(
                    $todayCompleted,
                    $todayTotal
                )
        ],


        /* WEEK */

        "week" => [

            "total" =>
                $weekTotal,

            "completed" =>
                $weekCompleted,

            "percentage" =>
                percentage(
                    $weekCompleted,
                    $weekTotal
                )
        ],


        /* MONTH */

        "month" => [

            "total" =>
                $monthTotal,

            "completed" =>
                $monthCompleted,

            "percentage" =>
                percentage(
                    $monthCompleted,
                    $monthTotal
                )
        ],


        /* STUDY HOURS */

        "studyHours" => [

            "hours" =>
                $studyHours,

            "minutes" =>
                $studyRemainingMinutes
        ],


        /* TODAY TASKS */

        "tasks" =>
            $formattedTasks
    ]);

    exit;
}
